set allow-duplicate-variables

# Optional modules: import? allows `just fetch` to work before .just/remote/ exists.
import? '.just/remote/go.just'
import? '.just/remote/md.just'
import? '.just/remote/just.just'

# No documentation site, so md formats every markdown file in the repository.
md_site_dir := ""

default:
    @just --list

# --- Fetch ---

# Fetch shared justfiles
fetch:
    mkdir -p .just/remote
    curl -sSfL {{ justfilesRepo }}/go/go.just -o .just/remote/go.just
    curl -sSfL {{ justfilesRepo }}/md/md.just -o .just/remote/md.just
    curl -sSfL {{ justfilesRepo }}/just/just.just -o .just/remote/just.just

# --- Top-level orchestration ---

# Install all dependencies
deps:
    just go-deps
    just go-mod

# Run all tests
test:
    just go-test

# Format, lint, and generate before committing
ready:
    just md-fmt
    just go-fmt
    just go-vet
    just just-fmt
